from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from pydantic import BaseModel
from src.core.config import settings

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="http://127.0.0.1:8002/api/v1/auth/login")

class CurrentUser(BaseModel):
    id: str

def get_current_user(token: str = Depends(oauth2_scheme)) -> CurrentUser:
    """Weryfikuje JWT i zwraca obiekt usera."""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        user_id = payload.get("sub")
        if user_id is None:
            raise credentials_exception
        return CurrentUser(id=user_id)
    except JWTError:
        raise credentials_exception

from fastapi import APIRouter, Depends, status
from sqlalchemy.ext.asyncio import AsyncSession
from typing import List

from src.db.database import get_db
from src.schemas.task import TaskCreate, TaskUpdate, TaskResponse, TaskReorder
from src.api.dependencies import get_current_user, CurrentUser
from src.crud.task_repo import TaskRepository

router = APIRouter(prefix="/api/v1/tasks", tags=["Tasks"])

# Funkcja pomocnicza tworząca instancję repozytorium (Dependency Injection)
def get_task_repo(db: AsyncSession = Depends(get_db)) -> TaskRepository:
    return TaskRepository(db)

@router.get("/", response_model=List[TaskResponse])
async def get_tasks(
    current_user: CurrentUser = Depends(get_current_user),
    repo: TaskRepository = Depends(get_task_repo)
):
    return await repo.get_user_tasks(current_user.id)

@router.post("/", response_model=TaskResponse, status_code=status.HTTP_201_CREATED)
async def create_task(
    task_in: TaskCreate,
    current_user: CurrentUser = Depends(get_current_user),
    repo: TaskRepository = Depends(get_task_repo)
):
    return await repo.create_task(current_user.id, task_in)

@router.put("/{task_id}", response_model=TaskResponse)
async def update_task(
    task_id: int,
    task_in: TaskUpdate,
    current_user: CurrentUser = Depends(get_current_user),
    repo: TaskRepository = Depends(get_task_repo)
):
    return await repo.update_task(current_user.id, task_id, task_in)

@router.delete("/{task_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_task(
    task_id: int,
    current_user: CurrentUser = Depends(get_current_user),
    repo: TaskRepository = Depends(get_task_repo)
):
    await repo.delete_task(current_user.id, task_id)

@router.post("/{task_id}/reorder", response_model=TaskResponse)
async def reorder_task(
    task_id: int,
    reorder_data: TaskReorder,
    current_user: CurrentUser = Depends(get_current_user),
    repo: TaskRepository = Depends(get_task_repo)
):
    """
    Endpoint wywoływany, gdy użytkownik "upuści" zadanie (Drag & Drop) we Flutterze.
    """
    return await repo.reorder_task(
        user_id=current_user.id,
        task_id=task_id,
        prev_id=reorder_data.prev_task_id,
        next_id=reorder_data.next_task_id
    )

from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from sqlalchemy.sql import func
from src.db.models import Task
from src.schemas.task import TaskCreate, TaskUpdate
from src.core.config import settings
from fastapi import HTTPException

class TaskRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_user_tasks(self, user_id: str):
        # Pobieramy zadania posortowane po pozycji (rosnąco)
        query = select(Task).filter(Task.user_id == user_id).order_by(Task.position.asc())
        result = await self.db.execute(query)
        return result.scalars().all()

    async def create_task(self, user_id: str, task_in: TaskCreate) -> Task:
        # Znajdź najwyższą pozycję obecnie w bazie dla tego usera
        query = select(func.max(Task.position)).filter(Task.user_id == user_id)
        result = await self.db.execute(query)
        max_pos = result.scalar()
        
        # Jeśli nie ma zadań, startujemy od 1000. W przeciwnym razie dodajemy 1000 do najwyższej.
        new_position = settings.POSITION_STEP if max_pos is None else max_pos + settings.POSITION_STEP

        db_task = Task(
            user_id=user_id,
            content=task_in.content,
            position=new_position
        )
        self.db.add(db_task)
        await self.db.commit()
        await self.db.refresh(db_task)
        return db_task

    async def update_task(self, user_id: str, task_id: int, task_in: TaskUpdate) -> Task:
        task = await self.get_task_by_id(user_id, task_id)
        
        if task_in.content is not None:
            task.content = task_in.content
        if task_in.is_completed is not None:
            task.is_completed = task_in.is_completed
            
        await self.db.commit()
        await self.db.refresh(task)
        return task

    async def delete_task(self, user_id: str, task_id: int):
        task = await self.get_task_by_id(user_id, task_id)
        await self.db.delete(task)
        await self.db.commit()

    async def reorder_task(self, user_id: str, task_id: int, prev_id: int | None, next_id: int | None) -> Task:
        """
        Główny algorytm obliczania nowej pozycji.
        """
        task_to_move = await self.get_task_by_id(user_id, task_id)
        
        prev_task = await self.get_task_by_id(user_id, prev_id) if prev_id else None
        next_task = await self.get_task_by_id(user_id, next_id) if next_id else None

        if prev_task and next_task:
            # Wstawienie pomiędzy dwa istniejące zadania (wyciągamy średnią)
            new_pos = (prev_task.position + next_task.position) / 2.0
        elif prev_task and not next_task:
            # Przeniesienie na sam dół listy
            new_pos = prev_task.position + settings.POSITION_STEP
        elif not prev_task and next_task:
            # Przeniesienie na samą górę listy
            new_pos = next_task.position / 2.0  # Średnia między 0 a pierwszym elementem
        else:
            # Jeśli front wyśle oba jako None (lista miała 1 element), nic nie robimy
            new_pos = task_to_move.position

        task_to_move.position = new_pos
        await self.db.commit()
        await self.db.refresh(task_to_move)
        return task_to_move

    async def get_task_by_id(self, user_id: str, task_id: int) -> Task:
        query = select(Task).filter(Task.id == task_id, Task.user_id == user_id)
        result = await self.db.execute(query)
        task = result.scalars().first()
        if not task:
            raise HTTPException(status_code=404, detail="Task not found")
        return task

from typing import AsyncGenerator
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from sqlalchemy.orm import declarative_base
from src.core.config import settings

# Używamy aiosqlite dla pełnej asynchroniczności
engine = create_async_engine(settings.DATABASE_URL, echo=False)

AsyncSessionLocal = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,
    autoflush=False
)

Base = declarative_base()

async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        yield session

from sqlalchemy import Column, Integer, String, Boolean, Float, DateTime, Index
from sqlalchemy.sql import func
from datetime import datetime, timezone
from src.db.database import Base

class Task(Base):
    __tablename__ = "tasks"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(String, index=True, nullable=False)
    
    content = Column(String(500), nullable=False)
    is_completed = Column(Boolean, default=False, nullable=False)
    
    # Magia sortowania: Float pozwala na wstawianie elementów "pomiędzy" bez zmiany pozostałych
    position = Column(Float, nullable=False)
    
    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))

    # Optymalizacja pod najczęstsze zapytanie: "daj zadania usera X, posortowane po pozycji"
    __table_args__ = (
        Index('ix_tasks_user_position', 'user_id', 'position'),
    )

from pydantic import BaseModel, Field, ConfigDict
from datetime import datetime
from typing import Optional

class TaskCreate(BaseModel):
    content: str = Field(..., min_length=1, max_length=500, description="Task content in English")

class TaskUpdate(BaseModel):
    content: Optional[str] = Field(None, min_length=1, max_length=500)
    is_completed: Optional[bool] = None

class TaskReorder(BaseModel):
    # ID zadań, pomiędzy które chcemy wstawić nasze zadanie (jedno z nich może być None, jeśli wstawiamy na samą górę/dół)
    prev_task_id: Optional[int] = None
    next_task_id: Optional[int] = None

class TaskResponse(BaseModel):
    id: int
    content: str
    is_completed: bool
    position: float
    created_at: datetime

    model_config = ConfigDict(from_attributes=True)
